### 1.1 分类题目计数 — `GET /api/quiz/categories` 增强

| 属性 | 值 |
|------|-----|
| 优先级 | P0 |
| 预估工时 | 0.5d |

**改动范围**：`app/services/quiz.py` 仅此一处。

**当前状态**：`list_categories()` 返回递归树形结构（`list[dict[str, Any]]`），`question_count` 固定为 0。路由用 `QuizCategoryTreeResponse`。

**方案**：在递归 `build()` 前批量查询题目计数，注入每个节点的 `question_count`。父节点计数 = 自身题目数 + 子节点题目数之和。

```python
from collections import defaultdict
from sqlalchemy import func

async def list_categories(self) -> list[QuizCategoryTreeResponse]:
    async with get_db_ctx() as db:
        cats = (await db.execute(
            select(QuizCategory).order_by(
                QuizCategory.parent_id.asc().nullsfirst(), QuizCategory.id.asc()
            )
        )).scalars().all()

        # 批量查询每个分类的题目数
        rows = (await db.execute(
            select(QuizQuestion.category_id, func.count())
            .group_by(QuizQuestion.category_id)
        )).all()
        direct = defaultdict(int, rows)

        # 递归构建树，自底向上聚合计数
        categories_by_id = {c.id: c for c in cats}
        children_by_parent: dict[int | None, list[QuizCategory]] = defaultdict(list)
        for c in cats:
            children_by_parent[c.parent_id].append(c)

        visited: set[int] = set()
        cat_counts: dict[int, int] = {}

        def build(category: QuizCategory, ancestors: set[int]) -> QuizCategoryTreeResponse | None:
            if category.id in ancestors:
                return None
            visited.add(category.id)
            children = [
                c for child in children_by_parent.get(category.id, [])
                if (c := build(child, ancestors | {category.id})) is not None
            ]
            child_total = sum(c.question_count for c in children)
            cat_counts[category.id] = direct.get(category.id, 0)
            return QuizCategoryTreeResponse(
                id=category.id,
                name=category.name,
                parent_id=category.parent_id,
                description=category.description,
                question_count=cat_counts[category.id] + child_total,
                children=children,
            )

        roots = [c for c in cats if c.parent_id is None or c.parent_id not in categories_by_id]
        tree = [c for r in roots if (c := build(r, set())) is not None]
        for c in cats:
            if c.id not in visited:
                node = build(c, set())
                if node is not None:
                    tree.append(node)
        return tree
```

**Service**：`app/services/quiz.py` 新增方法

**SQL 实现**（SQLAlchemy ORM）：

```python
# total_answers
select(func.count()).select_from(QuizRecord).where(QuizRecord.user_id == user_id)

# correct_answers
select(func.count()).select_from(QuizRecord).where(
    QuizRecord.user_id == user_id, QuizRecord.is_correct == True
)

# today_answers / today_correct
select(
    func.count(),
    func.sum(case((QuizRecord.is_correct == True, 1), else_=0)),
).select_from(QuizRecord).where(
    QuizRecord.user_id == user_id,
    func.date(QuizRecord.updated_at) == func.current_date(),
)
```

**缓存策略**：`total_questions` 用 Redis 缓存，题目新增/删除时通过 `AdminQuizService` 刷新。其余指标不缓存（实时计算）。

#### 1.3.2 抽题逻辑

```python
import random

async def _pick_questions(db, category_id, question_count, question_type) -> list[int]:
    base = select(QuizQuestion.id)
    if category_id is not None:
        base = base.where(QuizQuestion.category_id == category_id)
    if question_type is not None:
        base = base.where(QuizQuestion.question_type == question_type)
    rows = (await db.execute(base)).all()
    ids = [r[0] for r in rows]
    if len(ids) < question_count:
        raise BusinessException(f"题库仅 {len(ids)} 题，不足以抽取 {question_count} 题")
    return random.sample(ids, question_count)
```

**注意**：不直接 `SELECT ... ORDER BY RANDOM()`，避免全表扫描。先查出 id 列表再 Python `random.sample`。

### 1.4 分类维度进度 — `GET /api/quiz/progress`

| 属性 | 值 |
|------|-----|
| 优先级 | P1 |
| 预估工时 | 0.5d |

**实现**：对每个顶级分类，JOIN `quiz_question` + 子查询 `quiz_record`，用 `COUNT` + `SUM(CASE WHEN is_correct)` 聚合。

```python
async def get_progress(self, user_id: int, category_id: int | None):
    async with get_db_ctx() as db:
        base = select(QuizCategory).where(QuizCategory.parent_id.is_(None))
        if category_id is not None:
            base = base.where(QuizCategory.id == category_id)
        parents = (await db.execute(base)).scalars().all()
        results = []
        for p in parents:
            # collect all category_ids under this parent
            sub_ids = [p.id] + [
                c.id for c in
                (await db.execute(select(QuizCategory).where(QuizCategory.parent_id == p.id))).scalars().all()
            ]
            total = (await db.execute(
                select(func.count()).select_from(QuizQuestion).where(QuizQuestion.category_id.in_(sub_ids))
            )).scalar() or 0
            answered = (await db.execute(
                select(func.count(func.distinct(QuizRecord.question_id)))
                .where(QuizRecord.user_id == user_id, QuizRecord.question_id.in_(
                    select(QuizQuestion.id).where(QuizQuestion.category_id.in_(sub_ids))
                ))
            )).scalar() or 0
            correct = (await db.execute(
                select(func.count())
                .where(QuizRecord.user_id == user_id, QuizRecord.is_correct == True,
                       QuizRecord.question_id.in_(
                           select(QuizQuestion.id).where(QuizQuestion.category_id.in_(sub_ids))
                       ))
            )).scalar() or 0
            results.append({
                "category_id": p.id,
                "category_name": p.name,
                "total": total,
                "answered": answered,
                "correct": correct,
                "accuracy": round(correct / answered * 100, 1) if answered > 0 else 0,
            })
        return results
```

### 1.5 近期练习记录 — `GET /api/quiz/recent`

| 属性 | 值 |
|------|-----|
| 优先级 | P1 |
| 预估工时 | 0.5d |

**实现**：按 `updated_at DESC` 取最近 N 条 `quiz_record`，JOIN `quiz_question` 取题面。

```python
async def recent(self, user_id: int, limit: int = 10):
    async with get_db_ctx() as db:
        rows = (await db.execute(
            select(QuizRecord, QuizQuestion)
            .join(QuizQuestion, QuizRecord.question_id == QuizQuestion.id)
            .where(QuizRecord.user_id == user_id)
            .order_by(QuizRecord.updated_at.desc())
            .limit(limit)
        )).all()
        return [
            {**QuizQuestionResponse.model_validate(q).model_dump(),
             "user_answer": r.user_answer, "is_correct": r.is_correct}
            for r, q in rows
        ]
```
